# ----------------------------------------------------------------------
# 3. KERNEL — méta-règles épistémiques, produit des contraintes
# ----------------------------------------------------------------------

class ConstraintClass(Enum):
    SECURITE = 3       # priorité la plus haute
    EPISTEMIQUE = 2
    OPTIMISATION = 1   # priorité la plus basse


@dataclass
class Constraint:
    name: str
    cls: ConstraintClass
    detail: str


def kernel_decide(ctx: DecisionContext) -> list[Constraint]:
    """Le Kernel applique uniquement des méta-règles à partir du DecisionContext."""

    constraints: list[Constraint] = []

    # Vérification obligatoire
    if (
        ctx.truth_apt
        and ctx.verification == "required"
    ):
        constraints.append(Constraint(
            "verification_obligatoire",
            ConstraintClass.SECURITE,
            "Une affirmation vérifiable doit être vérifiée."
        ))

    # Sources externes
    if ctx.external_sources == "required":
        constraints.append(Constraint(
            "sources_min",
            ConstraintClass.EPISTEMIQUE,
            f"Au moins {CONFIG['min_sources_high_stakes']} sources indépendantes requises."
        ))

    # Consensus ≠ preuve (toujours actif)
    constraints.append(Constraint(
        "consensus_non_preuve",
        ConstraintClass.EPISTEMIQUE,
        "Un accord inter-modèles ne remplace jamais une source indépendante."
    ))

    # Traçabilité (toujours active)
    constraints.append(Constraint(
        "tracabilite",
        ConstraintClass.EPISTEMIQUE,
        "Toute conclusion doit rester justifiable et révisable."
    ))

    # Hypothèse factuelle
    if (
    	ctx.contains_hypothesis
    	and ctx.truth_apt
):
        constraints.append(Constraint(
            "checklist_hypothese",
            ConstraintClass.EPISTEMIQUE,
            "Appliquer la checklist VERITAS avant toute conclusion."
        ))

    # Optimisation
    if ctx.stakes_level == "low":
        constraints.append(Constraint(
            "cout_minimal",
            ConstraintClass.OPTIMISATION,
            "Un seul modèle suffit."
        ))

    return constraints

# ----------------------------------------------------------------------
# 4. CONFLICT RESOLVER — priorité par classe, pas par ordre d'écriture
# ----------------------------------------------------------------------

def resolve_conflicts(constraints: list[Constraint]) -> list[Constraint]:
    """
    Résout les conflits potentiels : une contrainte OPTIMISATION ne peut jamais
    annuler une contrainte SECURITE ou EPISTEMIQUE. Ici : si 'verification_obligatoire'
    est présente, on retire toute contrainte qui la contredirait implicitement.
    """
    names = {c.name for c in constraints}
    if "verification_obligatoire" in names:
        constraints = [c for c in constraints if c.name != "disable_verification"]
    # tri par priorité décroissante pour affichage / exécution
    return sorted(constraints, key=lambda c: c.cls.value, reverse=True)


# ----------------------------------------------------------------------
# 5. ORCHESTRATEUR — construit un pipeline satisfaisant les contraintes
# ----------------------------------------------------------------------

def build_pipeline(ctx: DecisionContext, constraints: list[Constraint]) -> dict:
    names = {x.name for x in constraints}
    pipeline = {
        "context": ctx.__dict__,
        "n_models": 1,
        "web_search_required": False,
        "min_independent_sources": 0,
        "mode_d": False,
        "verification": False,
        "constraints_applied": [f"[{x.cls.name}] {x.name}: {x.detail}" for x in constraints],
    }

    if "verification_obligatoire" in names:
        pipeline["verification"] = True
        pipeline["web_search_required"] = True

    if "sources_min" in names:
        pipeline["min_independent_sources"] = CONFIG["min_sources_high_stakes"]
        pipeline["n_models"] = 2
        pipeline["web_search_required"] = True

    if "cout_minimal" in names:
        pipeline["n_models"] = 1

    if "mode_d_obligatoire" in names:
        pipeline["mode_d"] = True
        pipeline["web_search_required"] = True

    if "checklist_hypothese" in names:
        pipeline["checklist_hypothese_annexe"] = True

    return pipeline


def run(prompt: str) -> dict:
        ctx = analyze_prompt(prompt)
	raw_constraints = kernel_decide(ctx)
	constraints = resolve_conflicts(raw_constraints)
	return build_pipeline(ctx, constraints)
